# The summary of a node: ts_subtree_summarize_children.
# A node is printed with all its fields only the first time it is dumped, so
# the script uses a copy of the node to see the summary before and after the
# change.
lang json
leaf r0 10 0:0:0 1:0:1 1 1 0 0 0
leaf r1 10 0:0:0 1:0:1 1 2 0 0 0
node r2 19 0 r0 r1
clone r3 r2
# The summary of a node that was summarized already does not change.
summarize r2
dump r2
dump r3
# The symbol of the node decides how the error cost of the children is
# counted, so the summary changes with it.
leaf r4 10 0:0:0 1:0:1 1 3 0 0 0
leaf r5 10 0:0:0 1:0:1 1 4 0 0 0
node r6 19 0 r4 r5
clone r7 r6
setsym r6 65535
summarize r6
dump r6
dump r7
# The repeat depth of a repetition: the children have the symbol of the node.
leaf r8 22 0:0:0 1:0:1 1 5 0 0 0
leaf r9 22 0:0:0 1:0:1 1 6 0 0 0
node r10 22 0 r8 r9
clone r11 r10
summarize r10
dump r10
dump r11
# A node with an error child is fragile on both sides.
error r12 64 0:0:0 1:0:1 1 7
leaf r13 10 0:0:0 1:0:1 1 8 0 0 0
node r14 19 0 r12 r13
clone r15 r14
summarize r14
dump r14
dump r15
